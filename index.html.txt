        // ============ توابع کمکی ============
        function getRandomInt(min, max) {
            return Math.floor(Math.random() * (max - min + 1)) + min;
        }

        function getRandomFloat(min, max, decimals = 2) {
            const num = Math.random() * (max - min) + min;
            return parseFloat(num.toFixed(decimals));
        }

        function shuffleArray(array) {
            const newArray = [...array];
            for (let i = newArray.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [newArray[i], newArray[j]] = [newArray[j], newArray[i]];
            }
            return newArray;
        }

        function formatNumber(num) {
            if (Number.isInteger(num)) return num.toString();
            return num.toFixed(3).replace(/\.?0+$/, '');
        }

        // ============ ایجاد سوال ============
        function generateQuestion() {
            // انتخاب دسته تصادفی
            const categories = Object.keys(unitCategories);
            const categoryKey = categories[Math.floor(Math.random() * categories.length)];
            const category = unitCategories[categoryKey];
            
            // انتخاب دو واحد متفاوت بر اساس سطح
            let fromIndex, toIndex;
            const maxUnitIndex = Math.min(level + 2, category.units.length - 1);
            
            do {
                fromIndex = getRandomInt(0, maxUnitIndex);
                toIndex = getRandomInt(0, maxUnitIndex);
            } while (fromIndex === toIndex);

            const fromUnit = category.units[fromIndex];
            const toUnit = category.units[toIndex];

            // تولید مقدار تصادفی بر اساس سطح
            let value;
            if (level <= 2) {
                value = getRandomInt(1, 100);
            } else if (level <= 4) {
                value = getRandomFloat(0.1, 500, getRandomInt(1, 2));
            } else {
                value = getRandomFloat(0.001, 1000, getRandomInt(1, 3));
            }

            // محاسبه پاسخ صحیح
            const baseValue = value * fromUnit.factor;
            const correctAnswer = baseValue / toUnit.factor;

            // تولید گزینه‌های اشتباه
            const wrongAnswers = new Set();
            const correctFormatted = formatNumber(parseFloat(correctAnswer.toFixed(3)));

            while (wrongAnswers.size < 3) {
                let wrongAnswer;
                const variation = correctAnswer * (0.1 + Math.random() * 0.9);
                
                if (Math.random() > 0.5) {
                    wrongAnswer = correctAnswer * (1 + (Math.random() * 0.5 + 0.1));
                } else {
                    wrongAnswer = correctAnswer * (1 - (Math.random() * 0.5 + 0.1));
                }
                
                // اطمینان از اینکه اشتباه است
                if (Math.abs(wrongAnswer - correctAnswer) > correctAnswer * 0.01 && wrongAnswer > 0) {
                    wrongAnswers.add(formatNumber(parseFloat(wrongAnswer.toFixed(3))));
                }
            }

            // ترکیب و مخلوط کردن گزینه‌ها
            const allOptions = shuffleArray([correctFormatted, ...Array.from(wrongAnswers)]);

            return {
                categoryKey,
                category,
                fromUnit,
                toUnit,
                value,
                correctAnswer: correctFormatted,
                options: allOptions,
                questionStr: `${formatNumber(value)} ${fromUnit.name} معادل چند ${toUnit.name} است؟`
            };
        }

        // ============ نمایش سوال ============
        function displayQuestion() {
            if (!gameActive) return;

            currentQuestion = generateQuestion();
            
            unitType.textContent = currentQuestion.category.name;
            questionText.textContent = currentQuestion.questionStr;
            targetUnit.textContent = '';
            feedback.textContent = '';
            feedback.className = 'feedback';
            
            optionsGrid.innerHTML = '';
            
            currentQuestion.options.forEach((option, index) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn';
                btn.textContent = option;
                btn.onclick = () => checkAnswer(option, btn);
                optionsGrid.appendChild(btn);
            });

            // به‌روزرسانی نوار پیشرفت
            const progress = (questionsInLevel % 5) * 20;
            progressFill.style.width = progress + '%';
        }

        // ============ بررسی پاسخ ============
        function checkAnswer(selected, btnElement) {
            if (!gameActive) return;

            totalQuestions++;
            const allButtons = optionsGrid.querySelectorAll('.option-btn');
            
            // غیرفعال کردن همه دکمه‌ها
            allButtons.forEach(btn => btn.style.pointerEvents = 'none');

            if (selected === currentQuestion.correctAnswer) {
                // پاسخ صحیح
                btnElement.classList.add('correct');
                correctAnswers++;
                combo++;
                
                // محاسبه امتیاز
                const basePoints = 10;
                const comboBonus = Math.min(combo - 1, 5) * 2;
                const timeBonus = Math.floor(timeLeft / 10);
                const points = basePoints + comboBonus + timeBonus;
                
                score += points;
                questionsInLevel++;
                
                feedback.textContent = `✅ عالی! +${points} امتیاز ${combo > 1 ? '🔥' : ''}`;
                feedback.className = 'feedback correct';
                
                if (combo >= 3) {
                    feedback.innerHTML += `<br><span class="combo-text">🔥 کامبو ×${combo}!</span>`;
                }
                
                // افزایش زمان
                timeLeft = Math.min(timeLeft + 2, 99);
                
                // تغییر شکل موشک بر اساس سطح
                updateRocketEmoji();
                
                // افزایش سطح
                if (questionsInLevel >= 5) {
                    level = Math.min(level + 1, 7);
                    questionsInLevel = 0;
                    levelIndicator.textContent = `🌍 مرحله ${level} - ${getLevelName(level)}`;
                    progressFill.style.width = '0%';
                }
                
            } else {
                // پاسخ اشتباه
                btnElement.classList.add('wrong');
                combo = 0;
                
                // نمایش پاسخ صحیح
                allButtons.forEach(btn => {
                    if (btn.textContent === currentQuestion.correctAnswer) {
                        btn.classList.add('correct');
                    }
                });
                
                feedback.textContent = `❌ اشتباه! پاسخ صحیح: ${currentQuestion.correctAnswer}`;
                feedback.className = 'feedback wrong';
                
                // کاهش زمان
                timeLeft = Math.max(timeLeft - 5, 0);
                
                // لرزش موشک
                rocketEmoji.style.animation = 'shake 0.5s ease';
                setTimeout(() => {
                    rocketEmoji.style.animation = 'float 2s ease-in-out infinite';
                }, 500);
            }

            // به‌روزرسانی امتیاز
            scoreEl.textContent = score;
            if (score > highScore) {
                highScore = score;
                highScoreEl.textContent = highScore;
                localStorage.setItem('unitConverterHighScore', highScore);
            }

            // سوال بعدی بعد از تاخیر کوتاه
            setTimeout(() => {
                if (gameActive) {
                    displayQuestion();
                }
            }, 1500);
        }

        function getLevelName(level) {
            const names = ['آسان', 'آسان', 'متوسط', 'متوسط', 'سخت', 'سخت', 'فضانورد حرفه‌ای'];
            return names[Math.min(level - 1, 6)];
        }

        function updateRocketEmoji() {
            const rockets = ['🚀', '🛸', '✈️', '🛩️', '🚁', '🛰️', '🌌'];
            rocketEmoji.textContent = rockets[Math.min(level - 1, 6)];
        }

        // ============ تایمر ============
        function startTimer() {
            timerInterval = setInterval(() => {
                timeLeft--;
                timerEl.textContent = timeLeft;
                
                // تغییر رنگ تایمر
                if (timeLeft <= 10) {
                    timerBox.style.borderColor = '#ff1744';
                    timerBox.style.boxShadow = '0 0 30px rgba(255, 23, 68, 0.6)';
                    timerEl.style.animation = 'pulse 0.5s infinite';
                } else if (timeLeft <= 20) {
                    timerBox.style.borderColor = '#ff9100';
                    timerBox.style.boxShadow = '0 0 20px rgba(255, 145, 0, 0.5)';
                }
                
                if (timeLeft <= 0) {
                    endGame();
                }
            }, 1000);
        }

        // ============ شروع بازی ============
        function startGame() {
            // ریست متغیرها
            score = 0;
            timeLeft = 60;
            combo = 0;
            totalQuestions = 0;
            correctAnswers = 0;
            level = 1;
            questionsInLevel = 0;
            gameActive = true;
            
            // ریست نمایش
            scoreEl.textContent = '0';
            highScoreEl.textContent = highScore;
            timerEl.textContent = '60';
            levelIndicator.textContent = '🌍 مرحله ۱ - آسان';
            feedback.textContent = '';
            feedback.className = 'feedback';
            progressFill.style.width = '0%';
            rocketEmoji.textContent = '🚀';
            rocketEmoji.style.animation = 'float 2s ease-in-out infinite';
            
            // ریست استایل تایمر
            timerBox.style.borderColor = '#ff5252';
            timerBox.style.boxShadow = '0 0 20px rgba(255, 82, 82, 0.4)';
            timerEl.style.animation = 'none';
            
            // مخفی کردن دکمه شروع
            startBtn.style.display = 'none';
            
            // شروع تایمر و نمایش سوال
            clearInterval(timerInterval);
            startTimer();
            displayQuestion();
        }

        // ============ پایان بازی ============
        function endGame() {
            gameActive = false;
            clearInterval(timerInterval);
            
            // ذخیره بیشترین امتیاز
            if (score > highScore) {
                highScore = score;
                localStorage.setItem('unitConverterHighScore', highScore);
                highScoreEl.textContent = highScore;
            }
            
            // نمایش صفحه پایان
            const accuracy = totalQuestions > 0 ? Math.round((correctAnswers / totalQuestions) * 100) : 0;
            
            questionText.textContent = '🎮 بازی تمام شد!';
            targetUnit.textContent = '';
            unitType.textContent = '';
            optionsGrid.innerHTML = '';
            feedback.textContent = '';
            progressFill.style.width = '100%';
            
            // نمایش آمار نهایی
            const statsHtml = `
                <div class="game-over">
                    <h2>🚀 ماموریت پایان یافت!</h2>
                    <div class="final-stats">
                        <div class="stat-item">
                            <div class="stat-value">${score}</div>
                            <div class="stat-label">امتیاز نهایی</div>
                        </div>
                        <div class="stat-item">
                            <div class="stat-value">${correctAnswers}/${totalQuestions}</div>
                            <div class="stat-label">پاسخ‌های صحیح</div>
                        </div>
                        <div class="stat-item">
                            <div class="stat-value">${accuracy}%</div>
                            <div class="stat-label">درصد accuracy</div>
                        </div>
                        <div class="stat-item">
                            <div class="stat-value">🏆 ${highScore}</div>
                            <div class="stat-label">رکورد</div>
                        </div>
                    </div>
                    <p style="color: #80deea; margin: 15px 0;">
                        ${accuracy >= 80 ? '🌟 عالی بود! تو یه فضانورد حرفه‌ای هستی!' : 
                          accuracy >= 50 ? '👍 خوب بود! با تمرین بیشتر می‌تونی بهتر بشی!' : 
                          '💪 به تمرین ادامه بده! هر بار قوی‌تر می‌شی!'}
                    </p>
                </div>
            `;
            
            optionsGrid.innerHTML = statsHtml;
            
            // نمایش دکمه شروع مجدد
            startBtn.textContent = '🔄 ماجراجویی دوباره';
            startBtn.style.display = 'block';
            startBtn.onclick = startGame;
            
            rocketEmoji.textContent = '🏁';
        }

        // ============ ایجاد ستاره‌های پس‌زمینه ============
        function createStars() {
            const container = document.getElementById('starsContainer');
            for (let i = 0; i < 100; i++) {
                const star = document.createElement('div');
                star.className = 'star';
                star.style.left = Math.random() * 100 + '%';
                star.style.top = Math.random() * 100 + '%';
                star.style.width = (Math.random() * 3 + 1) + 'px';
                star.style.height = star.style.width;
                star.style.animationDelay = Math.random() * 2 + 's';
                container.appendChild(star);
            }
            
            // ستاره‌های دنباله‌دار
            setInterval(() => {
                if (Math.random() < 0.3) {
                    const shootingStar = document.createElement('div');
                    shootingStar.className = 'shooting-star';
                    shootingStar.style.left = Math.random() * 100 + '%';
                    shootingStar.style.top = Math.random() * 50 + '%';
                    container.appendChild(shootingStar);
                    
                    setTimeout(() => {
                        shootingStar.remove();
                    }, 1000);
                }
            }, 3000);
        }
